"""
kali_realtime_sniffer.py

Same as your previous sniffer but with improved logging, error handling,
auto-retry for blocking, fallback to ufw, and optional auto-unblock TTL.

Run:
  sudo python3 kali_realtime_sniffer.py
"""

import threading
from collections import defaultdict, deque
from datetime import datetime
import json
import requests
import sys
import math
import subprocess
import logging
import shutil  # to check for ufw
from typing import Optional

from scapy.all import sniff, IP, TCP, UDP

# ---------- LOGGING SETUP ----------
logging.basicConfig(
    filename="kali_block_log.txt",
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
)
console = logging.StreamHandler()
console.setLevel(logging.INFO)
formatter = logging.Formatter("%(asctime)s [%(levelname)s] %(message)s")
console.setFormatter(formatter)
logging.getLogger().addHandler(console)

# ---------- CONFIG ----------
HOST = "http://192.168.1.9:5000"   # <-- change to your Flask host if needed
API = HOST + "/api/flow"
# If your server expects an API key header, set it here:
API_KEY = None  # e.g. "S0m3$tr0ngKey"
HEADERS = {"Content-Type": "application/json"}
if API_KEY:
    HEADERS["X-IDS-KEY"] = API_KEY

FLOW_TIMEOUT = 5         # seconds inactive to close flow
FLOW_MAX_AGE = 120       # force evaluate after this many seconds
WINDOW_SECS = 60         # sliding window secs for aggregated stats
SNAP_COUNT = 0           # 0 = continuous sniff
POST_TIMEOUT = 3         # requests timeout (seconds)
RETRY_POST = 2           # retries for POST to server

# Blocking retry/fallback config
BLOCK_RETRY = 2          # attempts to run iptables command
BLOCK_RETRY_BACKOFF = 1  # seconds (multiplied)
USE_UFW_FALLBACK = True  # use 'ufw' if iptables fails and ufw is available

# Auto-unblock config (0 = disabled). Set to seconds after which an IP is unblocked.
UNBLOCK_AFTER_SECS = 0   # e.g. 3600 to auto-unblock after 1 hour

# Safe hosts (won't be blocked)
SAFE_IPS = {"192.168.1.9", "192.168.119.103"}  # update with your actual host IPs

# ---------- FEATURES ----------
FEATURE_NAMES = [
    "pkSeqID", "stime", "flgs", "flgs_number", "proto_number",
    "pkts", "bytes", "state_number", "ltime", "seq", "dur",
    "mean", "stddev", "sum", "min", "max", "spkts", "dpkts",
    "sbytes", "dbytes", "rate", "srate", "drate",
    "TnBPSrcIP", "TnBPDstIP", "TnP_PSrcIP", "TnP_PDstIP",
    "TnP_PerProto", "TnP_Per_Dport", "AR_P_Proto_P_SrcIP",
    "AR_P_Proto_P_DstIP", "N_IN_Conn_P_DstIP", "N_IN_Conn_P_SrcIP",
    "AR_P_Proto_P_Sport", "AR_P_Proto_P_Dport",
    "Pkts_P_State_P_Protocol_P_DestIP",
    "Pkts_P_State_P_Protocol_P_SrcIP"
]

# ---------- data structures ----------
class Flow:
    def __init__(self, key, ts):
        self.key = key
        self.start = ts
        self.last = ts
        self.pkt_count = 0
        self.byte_count = 0
        self.pkt_sizes = []
        self.spkts = 0
        self.dpkts = 0
        self.sbytes = 0
        self.dbytes = 0
        self.flag_counts = defaultdict(int)
        self.state_number = 0

    def update(self, pkt_len, src_is_srcside=True, flags=None, ts=None):
        self.pkt_count += 1
        self.byte_count += pkt_len
        self.pkt_sizes.append(pkt_len)
        if src_is_srcside:
            self.spkts += 1
            self.sbytes += pkt_len
        else:
            self.dpkts += 1
            self.dbytes += pkt_len
        if flags:
            for ch in flags:
                self.flag_counts[ch] += 1
        if ts:
            self.last = ts

    def duration(self):
        return max(1e-6, self.last - self.start)

flows = {}
flows_lock = threading.Lock()

window_events = deque()
window_lock = threading.Lock()

seq_counter = 0

# blocking state
blocked_ips = set()
blocked_time = {}  # ip -> timestamp when blocked (for auto-unblock)

# helper to check ufw availability
UFW_AVAILABLE = shutil.which("ufw") is not None

# ---------- feature computations ----------
def proto_to_number(proto_str):
    if proto_str == "TCP":
        return 6
    if proto_str == "UDP":
        return 17
    try:
        return int(proto_str)
    except Exception:
        return 1

def compute_flow_features(flow, key):
    global seq_counter
    src, dst, sport, dport, proto = key
    seq_counter += 1

    pkt_count = flow.pkt_count
    byte_count = flow.byte_count
    duration = flow.duration()
    mean_pkt = sum(flow.pkt_sizes) / pkt_count if pkt_count else 0.0
    stddev_pkt = (sum((x-mean_pkt)**2 for x in flow.pkt_sizes) / pkt_count)**0.5 if pkt_count else 0.0
    sbytes = flow.sbytes
    dbytes = flow.dbytes
    spkts = flow.spkts
    dpkts = flow.dpkts
    min_pkt = min(flow.pkt_sizes) if flow.pkt_sizes else 0.0
    max_pkt = max(flow.pkt_sizes) if flow.pkt_sizes else 0.0
    sum_pkt = sum(flow.pkt_sizes)

    flag_total = sum(flow.flag_counts.values())
    flgs_number = len([k for k,v in flow.flag_counts.items() if v>0])
    flgs_percent = (flag_total / pkt_count * 100) if pkt_count else 0.0

    proto_number = proto_to_number(proto)
    state_number = 1 if ('F' in flow.flag_counts or 'R' in flow.flag_counts) else 0

    rate = byte_count / duration if duration > 0 else 0.0
    srate = sbytes / duration if duration > 0 else 0.0
    drate = dbytes / duration if duration > 0 else 0.0

    now = time.time()
    with window_lock:
        cutoff = now - WINDOW_SECS
        while window_events and window_events[0][0] < cutoff:
            window_events.popleft()

        total_bytes_window = sum(evt[4] for evt in window_events)
        total_pkt_window = len(window_events)

        bytes_per_src = defaultdict(int)
        bytes_per_dst = defaultdict(int)
        pkts_per_src = defaultdict(int)
        pkts_per_dst = defaultdict(int)
        proto_counts = defaultdict(int)
        sport_counts = defaultdict(int)
        dport_counts = defaultdict(int)
        conn_in_dst = defaultdict(int)
        conn_in_src = defaultdict(int)
        seen_src_dst_pairs = set()
        for evt in window_events:
            t, e_src, e_dst, e_proto, e_bytes, e_sport, e_dport = evt
            bytes_per_src[e_src] += e_bytes
            bytes_per_dst[e_dst] += e_bytes
            pkts_per_src[e_src] += 1
            pkts_per_dst[e_dst] += 1
            proto_counts[e_proto] += 1
            sport_counts[e_sport] += 1
            dport_counts[e_dport] += 1
            if (e_src, e_dst) not in seen_src_dst_pairs:
                conn_in_dst[e_dst] += 1
                conn_in_src[e_src] += 1
                seen_src_dst_pairs.add((e_src, e_dst))

        TnBPSrcIP = (bytes_per_src.get(src, 0) / total_bytes_window * 100) if total_bytes_window > 0 else 0.0
        TnBPDstIP = (bytes_per_dst.get(dst, 0) / total_bytes_window * 100) if total_bytes_window > 0 else 0.0

        TnP_PSrcIP = (pkts_per_src.get(src, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0
        TnP_PDstIP = (pkts_per_dst.get(dst, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0

        TnP_PerProto = (proto_counts.get(proto, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0
        TnP_Per_Dport = (dport_counts.get(dport, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0

        AR_P_Proto_P_SrcIP = (proto_counts.get(proto, 0) / (pkts_per_src.get(src, 1)) * 100) if pkts_per_src.get(src,0) > 0 else 0.0
        AR_P_Proto_P_DstIP = (proto_counts.get(proto, 0) / (pkts_per_dst.get(dst, 1)) * 100) if pkts_per_dst.get(dst,0) > 0 else 0.0

        N_IN_Conn_P_DstIP = conn_in_dst.get(dst, 0)
        N_IN_Conn_P_SrcIP = conn_in_src.get(src, 0)

        AR_P_Proto_P_Sport = (proto_counts.get(proto, 0) / (sport_counts.get(sport, 1)) * 100) if sport_counts.get(sport,0) > 0 else 0.0
        AR_P_Proto_P_Dport = (proto_counts.get(proto, 0) / (dport_counts.get(dport, 1)) * 100) if dport_counts.get(dport,0) > 0 else 0.0

        Pkts_P_State_P_Protocol_P_DestIP = (pkts_per_dst.get(dst, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0
        Pkts_P_State_P_Protocol_P_SrcIP = (pkts_per_src.get(src, 0) / total_pkt_window * 100) if total_pkt_window > 0 else 0.0

    feat = {
        "pkSeqID": float(seq_counter),
        "stime": float(flow.start % 100000),
        "flgs": float(round(flgs_percent, 2)),
        "flgs_number": float(flgs_number),
        "proto_number": float(proto_number),
        "pkts": float(pkt_count),
        "bytes": float(byte_count),
        "state_number": float(state_number),
        "ltime": float(flow.last % 100000),
        "seq": float(seq_counter),
        "dur": float(round(duration, 4)),
        "mean": float(round(mean_pkt, 4)),
        "stddev": float(round(stddev_pkt, 4)),
        "sum": float(round(sum_pkt, 4)),
        "min": float(min_pkt),
        "max": float(max_pkt),
        "spkts": float(spkts),
        "dpkts": float(dpkts),
        "sbytes": float(sbytes),
        "dbytes": float(dbytes),
        "rate": float(round(rate, 4)),
        "srate": float(round(srate, 4)),
        "drate": float(round(drate, 4)),
        "TnBPSrcIP": float(round(TnBPSrcIP, 4)),
        "TnBPDstIP": float(round(TnBPDstIP, 4)),
        "TnP_PSrcIP": float(round(TnP_PSrcIP, 4)),
        "TnP_PDstIP": float(round(TnP_PDstIP, 4)),
        "TnP_PerProto": float(round(TnP_PerProto, 4)),
        "TnP_Per_Dport": float(round(TnP_Per_Dport, 4)),
        "AR_P_Proto_P_SrcIP": float(round(AR_P_Proto_P_SrcIP, 4)),
        "AR_P_Proto_P_DstIP": float(round(AR_P_Proto_P_DstIP, 4)),
        "N_IN_Conn_P_DstIP": float(N_IN_Conn_P_DstIP),
        "N_IN_Conn_P_SrcIP": float(N_IN_Conn_P_SrcIP),
        "AR_P_Proto_P_Sport": float(round(AR_P_Proto_P_Sport, 4)),
        "AR_P_Proto_P_Dport": float(round(AR_P_Proto_P_Dport, 4)),
        "Pkts_P_State_P_Protocol_P_DestIP": float(round(Pkts_P_State_P_Protocol_P_DestIP, 4)),
        "Pkts_P_State_P_Protocol_P_SrcIP": float(round(Pkts_P_State_P_Protocol_P_SrcIP, 4))
    }

    ordered = {name: feat.get(name, 0.0) for name in FEATURE_NAMES}
    return ordered

# ---------- event recorder ----------
def add_window_event(ts, src, dst, proto, b, sport, dport):
    with window_lock:
        window_events.append((ts, src, dst, proto, b, sport, dport))

# ---------- blocking helpers ----------
def _run_cmd(cmd):
    """Run subprocess command, return (success, returncode, stdout, stderr)."""
    try:
        proc = subprocess.run(cmd, check=False, capture_output=True, text=True)
        return (proc.returncode == 0, proc.returncode, proc.stdout.strip(), proc.stderr.strip())
    except Exception as e:
        return (False, -1, "", str(e))

def _attempt_block_iptables(ip) -> bool:
    """Try to add an iptables DROP rule for ip. Returns True on success."""
    for attempt in range(1, BLOCK_RETRY + 1):
        cmd = ["sudo", "iptables", "-A", "INPUT", "-s", ip, "-j", "DROP"]
        success, rc, out, err = _run_cmd(cmd)
        if success:
            logging.info(f"iptables block succeeded for {ip} on attempt {attempt}")
            return True
        else:
            logging.warning(f"iptables attempt {attempt} failed for {ip}: rc={rc} err={err}")
            time.sleep(BLOCK_RETRY_BACKOFF * attempt)
    return False

def _attempt_block_ufw(ip) -> bool:
    """Try ufw deny from ip (if ufw available)."""
    if not UFW_AVAILABLE:
        logging.info("ufw not available; skipping ufw fallback.")
        return False
    cmd = ["sudo", "ufw", "deny", "from", ip]
    success, rc, out, err = _run_cmd(cmd)
    if success:
        logging.info(f"ufw deny succeeded for {ip}")
        return True
    else:
        logging.warning(f"ufw deny failed for {ip}: rc={rc} err={err}")
        return False

def block_ip(ip):
    """Block the given IP using iptables, with retries and optional ufw fallback."""
    if ip in SAFE_IPS:
        logging.info(f"Skipping block for SAFE IP {ip}")
        return
    if ip in blocked_ips:
        logging.info(f"IP {ip} already blocked; skipping.")
        return

    try:
        ok = _attempt_block_iptables(ip)
        if not ok and USE_UFW_FALLBACK:
            logging.info(f"iptables failed for {ip}; attempting ufw fallback.")
            ok = _attempt_block_ufw(ip)

        if ok:
            blocked_ips.add(ip)
            blocked_time[ip] = time.time()
            logging.info(f"Blocked IP: {ip}")
            print(f"[BLOCK] {ip} blocked")
        else:
            logging.error(f"Failed to block IP {ip} with both iptables and ufw.")
    except Exception as e:
        logging.error(f"Unexpected error in block_ip({ip}): {e}")

def _attempt_unblock_iptables(ip) -> bool:
    cmd = ["sudo", "iptables", "-D", "INPUT", "-s", ip, "-j", "DROP"]
    success, rc, out, err = _run_cmd(cmd)
    if success:
        logging.info(f"iptables unblock succeeded for {ip}")
        return True
    else:
        logging.warning(f"iptables unblock failed for {ip}: rc={rc} err={err}")
        return False

def _attempt_unblock_ufw(ip) -> bool:
    if not UFW_AVAILABLE:
        return False
    # ufw doesn't have a simple remove-by-from command in older versions; attempt to delete rule by exact syntax
    cmd = ["sudo", "ufw", "delete", "deny", "from", ip]
    success, rc, out, err = _run_cmd(cmd)
    if success:
        logging.info(f"ufw unblock succeeded for {ip}")
        return True
    else:
        logging.warning(f"ufw unblock failed for {ip}: rc={rc} err={err}")
        return False

def unblock_ip(ip):
    """Remove DROP rule for IP (best effort)."""
    if ip in SAFE_IPS:
        logging.info(f"Attempted to unblock SAFE IP {ip}; skipping.")
        return

    try:
        ok = _attempt_unblock_iptables(ip)
        if not ok and UFW_AVAILABLE:
            ok = _attempt_unblock_ufw(ip)

        if ok and ip in blocked_ips:
            blocked_ips.discard(ip)
        if ip in blocked_time:
            del blocked_time[ip]
        if ok:
            logging.info(f"Unblocked IP: {ip}")
            print(f"[UNBLOCK] {ip} unblocked")
        else:
            logging.error(f"Failed to unblock IP: {ip}")
    except Exception as e:
        logging.error(f"Unexpected error in unblock_ip({ip}): {e}")

# optional auto-unblock background thread
def auto_unblock_worker():
    if UNBLOCK_AFTER_SECS <= 0:
        return
    logging.info(f"Auto-unblock worker started (TTL={UNBLOCK_AFTER_SECS}s)")
    while True:
        now = time.time()
        to_unblock = []
        for ip, t in list(blocked_time.items()):
            if now - t >= UNBLOCK_AFTER_SECS:
                to_unblock.append(ip)
        for ip in to_unblock:
            logging.info(f"Auto-unblock TTL expired for {ip}; attempting to unblock.")
            unblock_ip(ip)
        time.sleep(max(10, UNBLOCK_AFTER_SECS // 6))

# ---------- flow sweeper ----------
def sweeper():
    while True:
        now = time.time()
        to_eval = []
        with flows_lock:
            for k, f in list(flows.items()):
                if (now - f.last) >= FLOW_TIMEOUT or (now - f.start) >= FLOW_MAX_AGE:
                    to_eval.append(k)
            for k in to_eval:
                flow = flows.pop(k, None)
                if flow:
                    feat = compute_flow_features(flow, k)
                    post_features(feat, k)
        time.sleep(1)

# ---------- post to Flask ----------
def post_features(feat_dict, key):
    src_ip = key[0]
    payload = feat_dict.copy()
    payload['src_ip'] = src_ip
    payload['dst_ip'] = key[1]

    # Log the attempt
    logging.info(f"POSTing features for {src_ip} -> {key[1]} | pkts={payload.get('pkts')} bytes={payload.get('bytes')}")

    for attempt in range(1, RETRY_POST + 1):
        try:
            r = requests.post(API, json=payload, headers=HEADERS, timeout=POST_TIMEOUT)
            if r.status_code == 200:
                logging.info(f"POST success for {src_ip} (attempt {attempt})")
                try:
                    resp = r.json()
                except Exception:
                    resp = {}
                    logging.warning("Failed to parse JSON from server response.")

                # Normalize server prediction field
                server_pred = None
                if isinstance(resp, dict):
                    # common keys
                    for k in ("prediction", "status", "alert"):
                        if k in resp:
                            server_pred = resp[k]
                            break
                if server_pred is None and isinstance(resp, str):
                    server_pred = resp

                server_pred_str = str(server_pred).lower() if server_pred is not None else ""

                # If server says block (match 'block' or 'blocked')
                if server_pred_str and ("block" in server_pred_str or "blocked" in server_pred_str):
                    logging.info(f"Server instructed to block {src_ip} (server_pred={server_pred})")
                    block_ip(src_ip)
                else:
                    logging.info(f"Server returned normal for {src_ip} (server_pred={server_pred})")
                return
            else:
                logging.warning(f"POST returned status {r.status_code} for {src_ip}: {r.text}")
        except requests.exceptions.RequestException as e:
            logging.error(f"POST attempt {attempt} failed for {src_ip}: {e}")
            # small backoff then retry
            time.sleep(0.5 * attempt)
        except Exception as e:
            logging.exception(f"Unexpected error during POST for {src_ip}: {e}")
            time.sleep(0.5 * attempt)

    logging.error(f"Failed to POST features for {src_ip} after {RETRY_POST} attempts")

# ---------- packet callback ----------
def pkt_callback(pkt):
    try:
        if IP not in pkt:
            return
        ip = pkt[IP]
        src = ip.src
        dst = ip.dst
        sport = 0
        dport = 0
        proto = "OTHER"
        flags = None
        if pkt.haslayer(TCP):
            proto = "TCP"
            sport = int(pkt[TCP].sport)
            dport = int(pkt[TCP].dport)
            flags = str(pkt[TCP].flags)
        elif pkt.haslayer(UDP):
            proto = "UDP"
            sport = int(pkt[UDP].sport)
            dport = int(pkt[UDP].dport)
        else:
            proto = str(ip.proto)

        plen = len(pkt)
        ts = time.time()
        key = (src, dst, sport, dport, proto)

        add_window_event(ts, src, dst, proto, plen, sport, dport)

        with flows_lock:
            if key not in flows:
                flows[key] = Flow(key, ts)
            # packet direction assumption: packet's src corresponds to flow src
            flows[key].update(pkt_len=plen, src_is_srcside=True, flags=flags, ts=ts)
    except Exception as e:
        logging.exception(f"pkt_callback error: {e}")

# ---------- main ----------
if __name__ == "__main__":
    logging.info("Starting flow sweeper thread...")
    t = threading.Thread(target=sweeper, daemon=True)
    t.start()

    # start auto-unblock thread if configured
    if UNBLOCK_AFTER_SECS > 0:
        t2 = threading.Thread(target=auto_unblock_worker, daemon=True)
        t2.start()

    logging.info("Starting packet capture... (Ctrl-C to stop)")
    try:
        sniff(prn=pkt_callback, store=False, count=SNAP_COUNT)
    except KeyboardInterrupt:
        logging.info("Stopping sniffing, finalizing remaining flows...")
        with flows_lock:
            for k, f in list(flows.items()):
                feat = compute_flow_features(f, k)
                post_features(feat, k)
        logging.info("Done. Exiting.")
        sys.exit(0)
